# Console script for the runtime smoke test. `@wait N` = wait N real seconds.
# @stop must always be the LAST command (commands are consumed asynchronously,
# the client stops reading stdin after shutdown begins).
@wait 45
gamerule doDaylightCycle false
gamerule randomTickSpeed 3
time set day
difficulty peaceful
@wait 2
# registries of the content mods
datapack list
@wait 2
# village / worldgen structure sets actually generated
locate structure ctov:large/village_plains_fortified
locate structure towns_and_towers:village_ocean
@wait 1
# block + item registries of cooking/farming/fishing mods (ids verified from jars)
setblock ~2 ~ ~ minecraft:furnace
setblock ~4 ~ ~ farmersdelight:cooking_pot
setblock ~6 ~ ~ farm_and_charm:crafting_bowl
setblock ~8 ~ ~ vinery:fermentation_barrel
give @a minecraft:stick
@wait 2
# entity registries of the animal mods
summon minecraft:cow ~ ~ ~2
summon minecraft:cow ~ ~ ~4
@wait 2
# block-entity / capability tick under load
forceload add ~ ~
@wait 4
debug start
@wait 5
debug stop
@wait 2
@stop
